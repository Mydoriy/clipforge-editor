⚡ Hướng dẫn thiết lập bảng & Auth RLS trên Supabase (Chạy 1 lần)

Vào Supabase Dashboard > SQL Editor, dán đoạn mã sau và ấn Run để tạo bảng bảo vệ bằng auth.uid():

CREATE TABLE IF NOT EXISTS public.clipforge_projects ( id TEXT PRIMARY KEY, user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid(), name TEXT NOT NULL, duration NUMERIC NOT NULL DEFAULT 15, fps INTEGER NOT NULL DEFAULT 30, width INTEGER NOT NULL DEFAULT 1080, height INTEGER NOT NULL DEFAULT 1920, data JSONB NOT NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL, updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL ); CREATE INDEX IF NOT EXISTS idx_clipforge_projects_user_updated ON public.clipforge_projects (user_id, updated_at DESC); ALTER TABLE public.clipforge_projects ENABLE ROW LEVEL SECURITY; CREATE POLICY "Users can view their own projects" ON public.clipforge_projects FOR SELECT TO authenticated USING ((select auth.uid()) = user_id); CREATE POLICY "Users can insert their own projects" ON public.clipforge_projects FOR INSERT TO authenticated WITH CHECK ((select auth.uid()) = user_id); CREATE POLICY "Users can update their own projects" ON public.clipforge_projects FOR UPDATE TO authenticated USING ((select auth.uid()) = user_id) WITH CHECK ((select auth.uid()) = user_id); CREATE POLICY "Users can delete their own projects" ON public.clipforge_projects FOR DELETE TO authenticated USING ((select auth.uid()) = user_id);
